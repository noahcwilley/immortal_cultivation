immortal_cultivation_add_xp_effect = {
	if = {
		limit = {
			has_trait = qi_condensation
		}

		add_trait_xp = {
			trait = qi_condensation
			track = $TRACK$
			value = $AMOUNT$
		}
	}
	else_if = {
		limit = {
			has_trait = foundation_establishment
		}

		add_trait_xp = {
			trait = foundation_establishment
			track = $TRACK$
			value = $AMOUNT$
		}
	}
}

immortal_cultivation_refresh_progress_effect = {
	if = {
		limit = {
			story_owner = {
				has_trait = qi_condensation
			}
		}
		set_variable = {
            name = cultivation_subrealm_xp
            value = {
				value = "story_owner.has_trait_xp(qi_condensation|subrealm)"
				divide = 10
				floor = yes
				add = 1
				max = 9
				if = {
				limit = {
					story_owner = {
						has_trait_xp = {
							trait = qi_condensation
							track = subrealm
							value >= 100
						}
					}
				}
				add = 1
}
			}
        }
		set_variable = {
			name = cultivation_quality_xp
			value = {
				value = "story_owner.has_trait_xp(qi_condensation|quality)"
				divide = 25
				floor = yes
				add = 1
			}
		}
	}
}

immortal_cultivation_initialize_qi_effect = {
	if = {
		limit = {
			has_trait = cultivation_realm
			NOT = { has_variable = immortal_cultivation_current_qi }
		}
		
		set_variable = {
			name = immortal_cultivation_current_qi
			value = immortal_cultivation_max_qi_value
		}
	}
}

immortal_cultivation_add_qi_effect = {
	if = {
		limit = {
			has_trait = cultivation_realm
		}

		immortal_cultivation_initialize_qi_effect = yes

		set_variable = {
			name = immortal_cultivation_current_qi
			value = {
				value = var:immortal_cultivation_current_qi
				add = {
					value = immortal_cultivation_max_qi_value
					divide = 5

					if = {
						limit = {
							has_character_flag = cultivation_seclusion
						}
						multiply = 1.5
					}
				}

				max = immortal_cultivation_max_qi_value
			}
		}
	}
}

immortal_cultivation_start_qi_recovery_effect = {
	immortal_cultivation_initialize_qi_effect = yes

	if = {
		limit = {
			has_trait = cultivation_realm
			NOT = {
				any_owned_story = {
					type = immortal_cultivation_qi_recovery_story
				}
			}
		}

		create_story = immortal_cultivation_qi_recovery_story
	}
}
